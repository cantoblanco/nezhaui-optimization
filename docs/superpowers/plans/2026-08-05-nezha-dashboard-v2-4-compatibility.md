> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `nezhaop.js` show live Detail and Network charts together on the official Nezha user frontend v2.4.2, preserve the old dual-mounted-panel behavior, and keep cycle-transfer rows working on current card layouts.

**Architecture:** Keep the distributable as one browser script. Exercise the real script in jsdom fixtures and retain static selector checks.

```js
const fs = require("node:fs");
const path = require("node:path");
const { JSDOM } = require("jsdom");

const source = fs.readFileSync(path.join(__dirname, "..", "..", "nezhaop.js"), "utf8");

function createDashboardDom({
  html,
  url = "https://nezha.example/",
  serviceData = { success: true, data: { cycle_transfer_stats: {} } },
} = {}) {
  const dom = new JSDOM(html || "<!doctype html><html><head></head><body><div id=\"root\"></div></body></html>", {
    pretendToBeVisual: true,
    runScripts: "outside-only",
    url,
  });
  const { window } = dom;
  window.requestAnimationFrame = (callback) => window.setTimeout(() => callback(Date.now()), 0);
  window.cancelAnimationFrame = (id) => window.clearTimeout(id);
  window.ResizeObserver = class ResizeObserver {
    constructor(callback) { this.callback = callback; }
    observe(target) { this.target = target; this.callback([{ target }]); }
    disconnect() { this.target = null; }
  };
  window.fetch = async () => ({ ok: true, json: async () => serviceData });
  window.TrafficScriptConfig = { interval: 60000, toggleInterval: 0 };
  return dom;
}

function loadScript(dom) {
  dom.window.eval(source);
  return dom.window.__NEZHAOP_RUNTIME__;
}

function settle(dom, milliseconds = 30) {
  return new Promise((resolve) => dom.window.setTimeout(resolve, milliseconds));
}

module.exports = { createDashboardDom, loadScript, settle };
```

```js
const detailState = { frame: null, frameReady: false, info: null, observer: null };

function getTabSection(info) {
  const tabs = info?.querySelector(".server-info-tab");
  let node = tabs;
  while (node && node.parentElement !== info) node = node.parentElement;
  return node?.parentElement === info ? node : null;
}

function getTabButton(name) {
  const tabs = Array.from(document.querySelectorAll('.server-info-tab [class*="cursor-pointer"]'));
  const pattern = name === "Network" ? /Network|网络/i : /Detail|详情/i;
  return tabs.find((tab) => pattern.test(tab.textContent || "")) ||
    (name === "Network" ? tabs[1] : tabs[0]) || null;
}

function getOverviewRoot(info) {
  const marker = info?.querySelector(".server-name");
  let node = marker;
  while (node && node.parentElement !== info) node = node.parentElement;
  return node?.parentElement === info ? node : null;
}

function getDetailRoot(info) {
  const marker = info?.querySelector(".server-charts");
  let node = marker;
  while (node && node.parentElement !== info) node = node.parentElement;
  return node?.parentElement === info ? node : null;
}

function isEmbeddedDetailView() {
  return new URL(location.href).searchParams.get(EMBED_QUERY_KEY) === "detail";
}

function postEmbeddedHeight(root) {
  const height = Math.max(1, Math.ceil(root.getBoundingClientRect().height || root.scrollHeight || 1));
  window.parent?.postMessage({ type: "nezhaop:detail-ready", height }, location.origin);
}

function initializeEmbeddedDetail(info) {
  document.documentElement.dataset.nezhaopEmbed = "detail";
  const overview = getOverviewRoot(info);
  const tabSection = getTabSection(info);
  const detailRoot = getDetailRoot(info);
  document.querySelector(".header-top")?.closest(".mx-auto")?.classList.add("nezhaop-embed-hidden");
  document.querySelector(".header-shell")?.classList.add("nezhaop-embed-hidden");
  document.querySelector(".server-footer")?.classList.add("nezhaop-embed-hidden");
  overview?.classList.add("nezhaop-embed-hidden");
  tabSection?.classList.add("nezhaop-embed-hidden");
  if (!detailRoot) return false;
  const resizeObserver = new ResizeObserver(() => postEmbeddedHeight(detailRoot));
  resizeObserver.observe(detailRoot);
  detailState.resizeObserver = resizeObserver;
  postEmbeddedHeight(detailRoot);
  return true;
}

function detailFrameUrl() {
  const url = new URL(location.href);
  url.searchParams.set(EMBED_QUERY_KEY, "detail");
  return url.href;
}

function ensureDetailFrame(info) {
  let frame = info.querySelector(`iframe[${DETAIL_FRAME_ATTRIBUTE}]`);
  if (!frame) {
    frame = document.createElement("iframe");
    frame.setAttribute(DETAIL_FRAME_ATTRIBUTE, "");
    frame.title = "Server detail charts";
    frame.src = detailFrameUrl();
    frame.hidden = true;
    frame.style.cssText = "border:0;width:100%;min-height:1px;overflow:hidden;";
    const tabSection = getTabSection(info);
    tabSection?.after(frame);
  }
  detailState.frame = frame;
  return frame;
}

function acceptEmbeddedMessage(event) {
  const frame = detailState.frame;
  if (!frame || event.origin !== location.origin || event.source !== frame.contentWindow) return;
  if (event.data?.type !== "nezhaop:detail-ready") return;
  const height = Math.max(1, Number(event.data.height) || 1);
  frame.style.height = `${height}px`;
  frame.hidden = false;
  detailState.frameReady = true;
  getTabSection(detailState.info)?.classList.add("nezhaop-detail-hidden");
  getTabButton("Network")?.click();
}

function initializeParentDetail(info) {
  detailState.info = info;
  const tabSection = getTabSection(info);
  const contentRoots = Array.from(info.children).filter((child) => {
    if (child === getOverviewRoot(info) || child === tabSection) return false;
    return child.tagName === "DIV" || child.tagName === "SECTION";
  });
  if (contentRoots.length >= 2) return initializeLegacyDetail(info, contentRoots);
  ensureDetailFrame(info);
  return true;
}

function startDetailController() {
  window.addEventListener("message", acceptEmbeddedMessage);
  const scan = () => {
    const info = document.querySelector(".server-info");
    if (!info) return;
    if (isEmbeddedDetailView()) initializeEmbeddedDetail(info);
    else initializeParentDetail(info);
  };
  scan();
  detailState.observer = new MutationObserver(scan);
  detailState.observer.observe(document.querySelector("#root") || document.body, { childList: true, subtree: true });
}
```

Replace `domObserver` with a root observer that scans immediately and observes `subtree: true`:

```js
const domObserver = (() => {
  const observers = new Map();
  let detector = null;

  function scan(onChange) {
    document.querySelectorAll(CARD_LIST_SELECTOR).forEach((section) => {
      if (observers.has(section)) return;
      const observer = new MutationObserver(() => onChange());
      observer.observe(section, { childList: true, subtree: true });
      observers.set(section, observer);
      onChange();
    });
    for (const [section, observer] of observers) {
      if (document.contains(section)) continue;
      observer.disconnect();
      observers.delete(section);
    }
  }

  function startSectionDetector(onChange) {
    const run = () => scan(onChange);
    detector = new MutationObserver(run);
    detector.observe(document.querySelector("#root") || document.body, { childList: true, subtree: true });
    run();
    return detector;
  }
```
